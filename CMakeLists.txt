cmake_minimum_required(VERSION 3.24)
project(PromptManager LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-Wall>)
include_directories(src)
add_executable(PromptManager WIN32 src/main.cpp src/app.rc src/shell/DesktopShell.cpp src/ui/LibraryWindow.cpp)
target_link_options(PromptManager PRIVATE -municode -static -s)
target_link_libraries(PromptManager PRIVATE shell32 advapi32 comctl32 gdiplus dwmapi gdi32 shcore ole32)
install(TARGETS PromptManager RUNTIME DESTINATION .)
enable_testing()
target_sources(PromptManager PRIVATE src/shell/EdgeButton.cpp)
function(app_test name)
  add_executable(${name}_tests tests/${name}_tests.cpp ${ARGN})
  target_link_options(${name}_tests PRIVATE -static)
  add_test(NAME ${name} COMMAND ${name}_tests)
endfunction()
app_test(placement)
find_path(SQLITE_HEADERS sqlite3.h REQUIRED)
find_file(SQLITE_STATIC libsqlite3.a HINTS "${SQLITE_HEADERS}/../lib" REQUIRED)
add_library(library STATIC src/library/PromptStore.cpp)
target_include_directories(library PUBLIC "${SQLITE_HEADERS}")
target_link_libraries(library PUBLIC "${SQLITE_STATIC}")
target_link_libraries(PromptManager PRIVATE library)
app_test(store)
app_test(query)
target_link_libraries(store_tests PRIVATE library)
target_link_libraries(query_tests PRIVATE library)
target_sources(PromptManager PRIVATE src/ui/PromptEditor.cpp src/platform/Clipboard.cpp)
app_test(clipboard src/platform/Clipboard.cpp)
target_sources(PromptManager PRIVATE src/ui/LibraryPaint.cpp src/ui/LibraryActions.cpp)
target_link_libraries(PromptManager PRIVATE msimg32 comdlg32 uxtheme)
app_test(gesture)
get_filename_component(UCRT_PREFIX "${SQLITE_HEADERS}" DIRECTORY)
foreach(dependency IN ITEMS sqlite3 gcc-libs crt winpthreads)
  install(DIRECTORY "${UCRT_PREFIX}/share/licenses/${dependency}/" DESTINATION "licenses/${dependency}" OPTIONAL)
endforeach()
app_test(artwork)
target_link_libraries(artwork_tests PRIVATE gdiplus)
app_test(import)
target_link_libraries(import_tests PRIVATE library)
